## 7.2 写推理脚本

执行：

```bash
cd ~/midterm_vla
source .venv/bin/activate 2>/dev/null || true

cat > scripts/01_smolvla_infer.py <<'PY'
import os
import json
from pathlib import Path

import torch
import numpy as np
from PIL import Image

MODEL_ID = os.environ.get("MODEL_ID", "lerobot/smolvla_base")
DATASET_ID = os.environ.get("DATASET_ID", "lerobot/svla_so101_pickplace")

ROOT = Path(__file__).resolve().parents[1]
FIG_DIR = ROOT / "figs"
LOG_DIR = ROOT / "logs"
FIG_DIR.mkdir(exist_ok=True, parents=True)
LOG_DIR.mkdir(exist_ok=True, parents=True)

def tensor_to_image(x):
    """把 LeRobot 里的图片 tensor 保存成普通 png。"""
    if not isinstance(x, torch.Tensor):
        return None

    x = x.detach().cpu()

    # 常见情况：C,H,W
    if x.ndim == 3 and x.shape[0] in [1, 3, 4]:
        x = x.permute(1, 2, 0)

    # 常见情况：H,W,C
    arr = x.numpy()

    # 如果是 0~1 浮点，就乘 255
    if arr.dtype != np.uint8:
        arr = arr.astype(np.float32)
        if arr.max() <= 1.5:
            arr = arr * 255.0
        arr = np.clip(arr, 0, 255).astype(np.uint8)

    if arr.ndim == 2:
        return Image.fromarray(arr)
    if arr.shape[-1] == 1:
        return Image.fromarray(arr[..., 0])
    if arr.shape[-1] >= 3:
        return Image.fromarray(arr[..., :3])
    return None

def shape_str(v):
    if isinstance(v, torch.Tensor):
        return f"Tensor(shape={tuple(v.shape)}, dtype={v.dtype}, device={v.device})"
    return f"{type(v).__name__}: {str(v)[:80]}"

def adapt_keys_for_smolvla(frame, policy):
    """
    目的：
    有些小数据集的图片 key 叫 observation.images.top / wrist，
    但 smolvla_base 可能期待 observation.images.camera1 / camera2 / camera3。
    这里为了 smoke test，把已有图片复制到模型期待的 key 上。
    这不是严格科学评测，只是为了证明推理链路能走通。
    """
    frame = dict(frame)

    expected_images = []
    try:
        for k in policy.config.input_features:
            if "image" in k or "images" in k:
                expected_images.append(k)
    except Exception:
        pass

    available_images = [k for k in frame.keys() if "image" in k or "images" in k]

    print("\n[INFO] available image keys:", available_images)
    print("[INFO] expected image keys:", expected_images)

    if available_images and expected_images:
        for i, exp in enumerate(expected_images):
            if exp not in frame:
                src = available_images[min(i, len(available_images) - 1)]
                frame[exp] = frame[src]
                print(f"[INFO] mapped {src} -> {exp}")

    return frame

print("=" * 80)
print("[STEP] Import LeRobot modules")

from lerobot.datasets.lerobot_dataset import LeRobotDataset
```

```bash
import os
from pathlib import Path

import torch

MODEL_ID = os.environ.get("MODEL_ID", "lerobot/smolvla_base")
DATASET_ID = os.environ.get("DATASET_ID", "lerobot/svla_so101_pickplace")

ROOT = Path(__file__).resolve().parents[1]
LOG_DIR = ROOT / "logs"
LOG_DIR.mkdir(exist_ok=True, parents=True)

from lerobot.datasets.lerobot_dataset import LeRobotDataset
from lerobot.policies.factory import make_pre_post_processors
from lerobot.policies.smolvla.modeling_smolvla import SmolVLAPolicy

def adapt_keys_for_smolvla(frame, policy):
    frame = dict(frame)

    expected_images = []
    try:
        for k in policy.config.input_features:
            if "image" in k or "images" in k:
                expected_images.append(k)
    except Exception:
        pass

    available_images = [k for k in frame.keys() if "image" in k or "images" in k]

    if available_images and expected_images:
        for i, exp in enumerate(expected_images):
            if exp not in frame:
                src = available_images[min(i, len(available_images) - 1)]
                frame[exp] = frame[src]

    return frame

def extract_loss(ret):
    if isinstance(ret, tuple):
        return ret[0]
    if isinstance(ret, dict):
        if "loss" in ret:
            return ret["loss"]
        for k, v in ret.items():
            if "loss" in k and torch.is_tensor(v):
                return v
    if torch.is_tensor(ret):
        return ret
    raise RuntimeError(f"Cannot extract loss from return type: {type(ret)}")

print("=" * 80)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("[INFO] device:", device)
print("[INFO] torch:", torch.__version__)
print("[INFO] hip:", getattr(torch.version, "hip", None))

print("[STEP] load policy")
policy = SmolVLAPolicy.from_pretrained(MODEL_ID).to(device)
policy.train()
print("[OK] policy loaded")

print("[STEP] build preprocess")
preprocess, postprocess = make_pre_post_processors(
    policy.config,
    MODEL_ID,
    preprocessor_overrides={"device_processor": {"device": str(device)}},
)
print("[OK] preprocess ready")

print("[STEP] load dataset")
dataset = LeRobotDataset(DATASET_ID)
frame = adapt_keys_for_smolvla(dict(dataset[0]), policy)
batch = preprocess(frame)
print("[OK] dataset sample ready")

print("[STEP] forward")
try:
    ret = policy.forward(batch)
except Exception as e1:
    print("[WARN] policy.forward(batch) failed:", repr(e1))
    print("[STEP] try policy(batch)")
    ret = policy(batch)

loss = extract_loss(ret)
print("[RESULT] loss:", float(loss.detach().cpu()))

print("[STEP] backward")
loss.backward()
print("[OK] backward succeeded")

# 统计有梯度的参数数量
grad_params = 0
for p in policy.parameters():
    if p.grad is not None:
        grad_params += 1

print("[RESULT] parameters with grad:", grad_params)

out_path = LOG_DIR / "02_one_backward_result.txt"
with open(out_path, "w", encoding="utf-8") as f:
    f.write(f"MODEL_ID={MODEL_ID}\n")
    f.write(f"DATASET_ID={DATASET_ID}\n")
    f.write(f"device={device}\n")
    f.write(f"loss={float(loss.detach().cpu())}\n")
    f.write(f"parameters_with_grad={grad_params}\n")

print("[OK] result saved to:", out_path)
print("=" * 80)
```

## 8.2 标准：跑 100 steps 小微调

先试这个命令：

```bash
cd ~/midterm_vla
source .venv/bin/activate 2>/dev/null || true

lerobot-train \
  --policy.path=lerobot/smolvla_base \
  --dataset.repo_id=lerobot/svla_so101_pickplace \
  --batch_size=4 \
  --steps=100 \
  --output_dir=outputs/train/smolvla_midterm_100steps \
  --job_name=smolvla_midterm_100steps \
  --policy.device=cuda \
  --policy.dtype=bfloat16 \
  --policy.push_to_hub=false \
  --wandb.enable=false \
  --policy.empty_cameras=1 \
  --rename_map='{"observation.images.top":"observation.images.camera1","observation.images.wrist":"observation.images.camera2"}' \
  2>&1 | tee logs/03_train_100steps.log
```

为什么加 `--rename_map` 和 `--policy.empty_cameras=1`？因为很多数据集的观测 key 和模型预训练配置里的 key 不一样。LeRobot 官方文档说明，rename map 可以把数据集里的 observation key 映射成模型期待的 key；如果数据集摄像头数量少于模型期待数量，可以设置 `empty_cameras` 让模型自动用 masked dummy camera 忽略缺失视角。该机制支持 SmolVLA。([Hugging Face][10])

如果上面命令报错，把 `--policy.dtype=bfloat16` 删除再跑：

```bash
lerobot-train \
  --policy.path=lerobot/smolvla_base \
  --dataset.repo_id=lerobot/svla_so101_pickplace \
  --batch_size=2 \
  --steps=100 \
  --output_dir=outputs/train/smolvla_midterm_100steps \
  --job_name=smolvla_midterm_100steps \
  --policy.device=cuda \
  --policy.push_to_hub=false \
  --wandb.enable=false \
  --policy.empty_cameras=1 \
  --rename_map='{"observation.images.top":"observation.images.camera1","observation.images.wrist":"observation.images.camera2"}' \
  2>&1 | tee logs/03_train_100steps_retry.log
```

如果还是报显存错误，把 batch size 改成 1：

```bash
--batch_size=1
```

如果还是不行，不要死磕。你已经有 `forward + backward` 的保底成果。

## 中期报告正文草稿

### （一）立项背景

随着具身智能和机器人基础模型的发展，机器人逐渐从传统的单任务、规则式控制，走向基于视觉、语言和动作统一建模的通用操作策略。双臂机器人相比单臂机器人具有更强的协作能力和操作灵活性，能够完成搬运、递交、叠放、装配等复杂任务。然而，双臂操作对感知、规划、控制和协同动作生成提出了更高要求，尤其在真实场景中还会受到物体形状、背景、光照、遮挡、语言指令变化等因素影响。

目前，训练高泛化能力的机器人策略通常依赖大量高质量数据。真实机器人数据采集成本高、周期长、设备要求高，难以在本科生项目周期内大规模完成。因此，仿真数据生成和域随机化成为提升机器人策略鲁棒性的有效方向。RoboTwin 2.0 提供了面向双臂操作的仿真数据生成和 Benchmark 框架，能够通过自动专家代码生成、仿真反馈修正和强域随机化生成大规模专家轨迹。另一方面，SmolVLA 提出了一种轻量级视觉-语言-动作模型，将多视角图像、自然语言指令和机器人状态映射为连续动作序列，具有较低训练和推理成本，更适合端侧部署方向。

因此，本项目围绕“端侧可部署的双臂操作算法”展开，计划以 RoboTwin 2.0 作为双臂任务数据源和评测平台，以 SmolVLA 作为轻量 VLA 策略基线，探索仿真数据驱动的双臂操作策略训练与部署方法。

---

### （二）项目研究内容及实施方案

本项目研究内容主要包括三个方面。

第一，开展双臂操作和视觉-语言-动作模型的文献调研。重点阅读 RoboTwin 2.0 和 SmolVLA 两篇论文，理解双臂操作数据生成、域随机化、VLA 模型架构、action chunk、flow matching、异步推理等关键概念，形成项目技术路线。

第二，完成轻量级 VLA 策略的最小复现。中期阶段优先选择 SmolVLA 作为主复现对象，基于 Hugging Face LeRobot 框架完成环境配置、模型加载、数据读取和动作预测。具体流程包括：安装 LeRobot 与 SmolVLA 依赖，加载 `lerobot/smolvla_base` 预训练模型，读取 LeRobot 格式机器人数据样本，调用 `select_action` 输出连续动作张量，并进一步尝试单步 forward/backward 或小规模训练，验证训练链路可运行。

第三，完成 RoboTwin 2.0 数据平台的可行性分析和轻量思想复现。由于 RoboTwin 2.0 完整仿真对 Linux + NVIDIA/CUDA 环境依赖较强，而当前可用服务器主要为 AMD ROCm 环境，因此中期阶段暂不进行完整 RoboTwin 仿真评测。我们将 RoboTwin 2.0 作为后续双臂任务数据源和 Benchmark，重点调研其数据格式、任务设置和域随机化机制，并通过图像扰动和语言指令改写构造轻量 domain randomization demo，为后续使用 RoboTwin 数据增强策略鲁棒性做准备。

---

### （三）项目实施进展及初步成果

项目立项以来，团队已完成以下工作。

第一，完成了 RoboTwin 2.0 和 SmolVLA 两篇论文的初步阅读和技术路线梳理。我们明确了两篇论文在项目中的分工：RoboTwin 2.0 主要提供双臂操作仿真任务、专家轨迹、域随机化和标准评测协议；SmolVLA 主要提供轻量级 VLA 策略模型，用于实现从视觉、语言和机器人状态到动作输出的端到端映射。

第二，完成了实验环境选择与可行性评估。团队目前拥有 MacBook Pro M1 Max 本地设备、AMD ROCm 大显存服务器和一台 16GB 异构加速卡服务器。经过调研，MacBook 主要用于文献阅读、报告撰写和 PPT 制作；AMD ROCm 服务器用于 SmolVLA 的 PyTorch 推理和小规模训练；16GB 异构加速卡平台由于环境不熟悉、显存较小，暂不作为中期主要实验平台。RoboTwin 2.0 完整仿真评测将放在后续 NVIDIA CUDA 环境中完成。

第三，完成了 SmolVLA 的最小推理链路复现。我们基于 LeRobot 框架安装了 SmolVLA 相关依赖，成功加载 `lerobot/smolvla_base` 预训练模型，并读取 LeRobot 数据集样本，调用 `select_action` 输出连续动作张量。该实验验证了图像、状态、语言输入经过 VLA 模型生成动作的基本流程。

第四，初步验证了训练链路。我们进一步尝试在真实 LeRobot 数据样本上进行 forward 计算 loss，并执行 backward，验证模型的训练管线能够运行。在时间允许情况下，进行了 100 steps 级别的小规模微调实验，用于观察训练日志和 loss 变化。由于中期阶段目标是验证流程可行性，因此暂未追求最终策略成功率。

第五，完成了 RoboTwin 2.0 域随机化思想的轻量复现。我们基于机器人观测图像构造了亮度变化、对比度变化、噪声扰动、遮挡扰动、背景色调扰动等图像增强，并生成多种中文和英文任务指令改写，用于模拟 RoboTwin 2.0 中视觉和语言层面的 domain randomization。该部分作为中期阶段的探索性成果，为后续仿真到真实迁移和策略鲁棒性提升提供基础。

目前遇到的主要困难包括：RoboTwin 2.0 完整仿真链路对硬件和依赖环境要求较高，当前 AMD ROCm 环境不适合直接运行完整 CUDA 仿真；RoboTwin 全量数据规模较大，当前硬盘空间和中期时间限制下不适合完整下载；SmolVLA 与不同 LeRobot 数据集之间可能存在摄像头 key 不匹配问题，需要通过 rename map 或配置修改解决。

下一阶段计划包括：申请或租用 NVIDIA CUDA 环境，优先完成 RoboTwin 2.0 单任务 quick check；选取 1～2 个典型双臂任务进行小规模训练和评测；尝试将 RoboTwin 数据与 SmolVLA 训练流程对接；进一步探索面向端侧部署的模型压缩、量化和异步推理方法。

---

### （四）结题预期目标

结题时计划形成以下成果。

第一，完成一套从双臂仿真数据到轻量 VLA 策略训练的实验流程，包括数据读取、模型加载、策略训练和基本评测。

第二，完成至少一个双臂操作任务的小规模复现实验。任务可优先选择 RoboTwin 2.0 中相对基础的叠放、抓取、搬运类任务，例如 `stack_blocks_two`、`pick_dual_bottles` 或类似任务。
