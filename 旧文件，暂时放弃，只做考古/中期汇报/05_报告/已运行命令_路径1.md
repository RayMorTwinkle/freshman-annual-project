## 前置步骤：DSW 2222 端口 SSH 配置

> DSW 的 22 端口已被平台占用，需自行在 2222 端口启动 sshd。MacBook 公钥 `macbook-dsw`。

```bash

EOF

chmod 600 /root/.ssh/authorized_keys

cat /root/.ssh/authorized_keys
```

### 前置 2：安装 SSH 服务和工具

```bash
apt-get update
apt-get install -y openssh-server procps net-tools
```

### 前置 3：创建 2222 端口 sshd 配置

```bash
mkdir -p /run/sshd

cat > /tmp/sshd_dsw_2222_config <<'EOF'
Port 2222
ListenAddress 0.0.0.0
PermitRootLogin prohibit-password
PubkeyAuthentication yes
PasswordAuthentication no
ChallengeResponseAuthentication no
UsePAM no
AuthorizedKeysFile .ssh/authorized_keys
PidFile /tmp/sshd_dsw_2222.pid
LogLevel VERBOSE
Subsystem sftp /usr/lib/openssh/sftp-server
EOF

/usr/sbin/sshd -f /tmp/sshd_dsw_2222_config -t
```

### 前置 4：启动 2222 端口 SSH

```bash
nohup /usr/sbin/sshd -D -e -f /tmp/sshd_dsw_2222_config > /tmp/sshd_2222.log 2>&1 &

sleep 2

ps -ef | grep '[s]shd' || true

cat /tmp/sshd_2222.log
```

✅ 输出：`Server listening on 0.0.0.0 port 2222.`

### 3.6 修复 Python 3.12 兼容性

lerobot 0.5.1 的 GR00T policy 在 Python 3.12 下有 dataclass 字段顺序问题。需要注释掉三处：

```bash
PKG=/mnt/workspace/midterm_vla/.venv/lib/python3.12/site-packages/lerobot

# 1) policies/__init__.py：注释 groot 导入
sed -i 's/^from \.groot/# from .groot/' $PKG/policies/__init__.py

# 2) policies/factory.py：注释 groot 导入 + 把两个引用改成 skip
sed -i 's/from lerobot.policies.groot/# from lerobot.policies.groot/' $PKG/policies/factory.py
sed -i 's/if isinstance(policy_cfg, GrootConfig):/if False:/' $PKG/policies/factory.py
sed -i 's/elif policy_type == "groot":/elif False:  # groot disabled/' $PKG/policies/factory.py
sed -i 's/elif name == "groot":/elif False:  # groot disabled/' $PKG/policies/factory.py
sed -i 's/return GrootConfig(\*\*kwargs)/raise NotImplementedError()/' $PKG/policies/factory.py
sed -i 's/return GrootPolicy/raise NotImplementedError()/' $PKG/policies/factory.py
```

## 第 6 步：推理链路复现（模型加载 → 数据读取 → select_action）

### 6.1 写入推理脚本

```bash
cat > /mnt/workspace/midterm_vla/scripts/01_smolvla_infer.py << 'ENDOFSCRIPT'
import os, json
from pathlib import Path
import torch, numpy as np
from PIL import Image

MODEL_ID = "lerobot/smolvla_base"
DATASET_ID = "lerobot/svla_so101_pickplace"
ROOT = Path(__file__).resolve().parents[1]
FIG_DIR = ROOT / "figs"
LOG_DIR = ROOT / "logs"
FIG_DIR.mkdir(exist_ok=True, parents=True)
LOG_DIR.mkdir(exist_ok=True, parents=True)

def tensor_to_image(x):
    if not isinstance(x, torch.Tensor):
        return None
    x = x.detach().cpu()
    if x.ndim == 3 and x.shape[0] in [1, 3, 4]:
        x = x.permute(1, 2, 0)
    arr = x.numpy()
    if arr.dtype != np.uint8:
        arr = arr.astype(np.float32)
        if arr.max() <= 1.5:
            arr = arr * 255.0
        arr = np.clip(arr, 0, 255).astype(np.uint8)
    if arr.ndim == 2:
        return Image.fromarray(arr)
    if arr.shape[-1] == 1:
        return Image.fromarray(arr[..., 0])
    if arr.shape[-1] >= 3:
        return Image.fromarray(arr[..., :3])
    return None

print("=" * 60)
print("[STEP] Imports")
from lerobot.datasets.lerobot_dataset import LeRobotDataset
from lerobot.policies.factory import make_pre_post_processors
from lerobot.policies.smolvla.modeling_smolvla import SmolVLAPolicy
print("[OK]")

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"[INFO] device: {device}, torch: {torch.__version__}")

print("[STEP] Load policy")
policy = SmolVLAPolicy.from_pretrained(MODEL_ID).to(device).eval()
n = sum(p.numel() for p in policy.parameters())
print(f"[OK] {n/1e6:.1f}M params")

print("[STEP] Build preprocess/postprocess")
preprocess, postprocess = make_pre_post_processors(
    policy.config, MODEL_ID,
    preprocessor_overrides={"device_processor": {"device": str(device)}},
)
print("[OK]")

print(f"[STEP] Load dataset: {DATASET_ID}")
# ⚠️ 关键：AMD ROCm 环境 torchcodec 缺少 libavutil.so，必须用 pyav 后端
dataset = LeRobotDataset(DATASET_ID, video_backend="pyav")
print(f"[OK] {len(dataset)} samples")

frame = dict(dataset[0])
print("[INFO] frame keys:", list(frame.keys()))

img_keys = [k for k in frame if "image" in k]
if img_keys:
    img = tensor_to_image(frame[img_keys[0]])
    if img:
        p = FIG_DIR / "sample_observation.png"
        img.save(p)
        print(f"[OK] saved {p}")

avail = [k for k in frame if "image" in k]
exp_list = [k for k in policy.config.input_features if "image" in k]
if avail and exp_list:
    for i, exp in enumerate(exp_list):
        if exp not in frame:
            src = avail[min(i, len(avail)-1)]
            frame[exp] = frame[src]
            print(f"[INFO] mapped {src} -> {exp}")

print("[STEP] Preprocess")
batch = preprocess(frame)
print("[OK]")

print("[STEP] select_action")
with torch.inference_mode():
    out = policy.select_action(batch)
print(f"[OK] output type: {type(out)}")

if isinstance(out, torch.Tensor):
    print(f"[RESULT] action shape: {tuple(out.shape)}, dtype: {out.dtype}")
    print(f"[RESULT] first values: {out.detach().cpu().flatten()[:8].tolist()}")
elif isinstance(out, dict):
    for k, v in out.items():
        if isinstance(v, torch.Tensor):
            print(f"[RESULT] {k}: shape={tuple(v.shape)}")
else:
    print(f"[RESULT] {out}")

print("=" * 60)
print("DONE - inference pipeline complete!")
ENDOFSCRIPT
```
